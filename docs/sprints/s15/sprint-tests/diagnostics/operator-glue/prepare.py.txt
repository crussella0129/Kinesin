"""Disposable live-workload setup and source/binary provenance; no inference."""

import hashlib
import json
import shutil
import socket
import subprocess
from datetime import datetime, timezone
from pathlib import Path

repo = Path(__file__).resolve().parents[2]
lab = repo / "target/s15-live"


def digest(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def port():
    with socket.socket() as reserve:
        reserve.bind(("127.0.0.1", 0))
        return reserve.getsockname()[1]


runtime = {
    "model_port": port(),
    "relay_port": port(),
    "alias": "s15-causal-qwen25",
    "runtime_path": str(Path.home() / ".cargo/bin/runtime/llama-server.exe"),
    "model_path": str(Path.home() / ".cargo/bin/models/qwen2.5-coder-7b-instruct-q4_k_m.gguf"),
    "seed": 0,
}
runtime_file = lab / "runtime.json"
if runtime_file.exists():
    raise SystemExit("runtime already frozen; do not overwrite")
for label in ["runtime", "model"]:
    runtime[label + "_sha256"] = digest(Path(runtime[label + "_path"]))
runtime_file.write_text(json.dumps(runtime, indent=2), encoding="utf-8")
template = (repo / "target/s14-live/attempt-7/control/kinesin.toml").read_text(encoding="utf-8")
template = template.split("[managed_model]")[0]
template = template.replace('base_url = "http://127.0.0.1:8080"', f'base_url = "http://127.0.0.1:{runtime["relay_port"]}"')
template = template.replace('model_id = "local"', f'model_id = "{runtime["alias"]}"')
source_files = list((repo / "src").rglob("*.rs")) + [repo / "Cargo.toml", repo / "Cargo.lock"]
sources = {str(p.relative_to(repo)).replace("\\", "/"): digest(p) for p in sorted(source_files)}
for number, adapter, executable in [
    (1, 5, repo / "target/s14-live/attempt-7/control/kinesin-live.exe"),
    (2, 6, repo / "target/debug/kinesin.exe"),
]:
    call = lab / f"call-{number:02d}"
    call.mkdir()
    shutil.copytree(lab / "fixtures/order-base", call / "app")
    (call / "control").mkdir()
    (call / "evidence").mkdir()
    profile = template
    if adapter == 6:
        profile = profile.replace('stream = false', 'stream = false\naction_protocol = "structured_ordered_v1"')
    config = call / "control/kinesin.toml"
    config.write_text(profile, encoding="utf-8")
    prompt = lab / "fixtures/prompts/pair-1.txt"
    first = call / "evidence/expected-first.json"
    subprocess.run([str(repo / "target/debug/examples/s15-prepare-request.exe"), str(config), str(prompt), str(adapter), str(first)], check=True)
    freeze = {
        "slot": number, "adapter": adapter, "frozen_utc": datetime.now(timezone.utc).isoformat(),
        "config": str(config), "config_sha256": digest(config),
        "prompt": str(prompt), "prompt_sha256": digest(prompt),
        "executable": str(executable), "binary_sha256": digest(executable),
        "expected_first_sha256": digest(first), "runtime": runtime,
        "current_source_files": sources,
        "control_source_note": "slot1 uses retained S14 attempt7 binary 0c26...; slot2 current candidate; exact first wire comparison isolates ordering",
        "seed": {str(p.relative_to(call / "app")): digest(p) for p in sorted((call / "app").rglob("*")) if p.is_file()},
    }
    (call / "evidence/freeze.json").write_text(json.dumps(freeze, indent=2), encoding="utf-8")
old = json.loads((lab / "call-01/evidence/expected-first.json").read_bytes())
new = json.loads((lab / "call-02/evidence/expected-first.json").read_bytes())
if old != new:
    raise SystemExit("first bodies differ semantically; refuse dispatch")
for item in old["response_format"]["json_schema"]["schema"]["oneOf"]:
    properties = item["properties"]
    keys = ["kind", "name", "arguments"] if properties["kind"]["const"] == "tool" else ["kind", "text"]
    item["properties"] = {key: properties[key] for key in keys}
reordered = json.dumps(old, separators=(",", ":"), ensure_ascii=False).encode()
candidate = (lab / "call-02/evidence/expected-first.json").read_bytes()
if reordered != candidate:
    raise SystemExit("additional byte differences beyond schema order; refuse dispatch")
(lab / "pair-1-wire-audit.json").write_text(json.dumps({"only_schema_property_order_differs": True, "first_body_semantically_equal": True, "all_tool_properties": ["kind", "name", "arguments"], "answer_properties": ["kind", "text"]}, indent=2), encoding="utf-8")
print(json.dumps({"runtime": runtime, "pair_prepared": True}, indent=2))
